# Reference Ceph CRUSH map with rack-isolated Straw2 replicated and
# erasure-coded placement rules.
# Compile and test with:
#   crushtool -c crush-rack-map.txt -o crush.bin
#   crushtool -i crush.bin --test --show-utilization --rule 0 --num-rep 3

tunable choose_local_tries 0
tunable choose_local_fallback_tries 0
tunable choose_total_tries 50
tunable chooseleaf_descend_once 1
tunable chooseleaf_vary_r 1
tunable straw_calc_version 1

type 0 osd
type 1 host
type 2 rack
type 3 datacenter
type 4 root

device 0 osd.0 class nvme
device 1 osd.1 class nvme
device 2 osd.2 class nvme
device 3 osd.3 class nvme
device 4 osd.4 class nvme
device 5 osd.5 class nvme

host host-a1 { id -101 alg straw2 hash 0 item osd.0 weight 1.000 }
host host-a2 { id -102 alg straw2 hash 0 item osd.1 weight 1.000 }
host host-b1 { id -201 alg straw2 hash 0 item osd.2 weight 1.000 }
host host-b2 { id -202 alg straw2 hash 0 item osd.3 weight 1.000 }
host host-c1 { id -301 alg straw2 hash 0 item osd.4 weight 1.000 }
host host-c2 { id -302 alg straw2 hash 0 item osd.5 weight 1.000 }

rack rack-a { id -10 alg straw2 hash 0 item host-a1 weight 1.000 item host-a2 weight 1.000 }
rack rack-b { id -20 alg straw2 hash 0 item host-b1 weight 1.000 item host-b2 weight 1.000 }
rack rack-c { id -30 alg straw2 hash 0 item host-c1 weight 1.000 item host-c2 weight 1.000 }

root default {
    id -1
    alg straw2
    hash 0
    item rack-a weight 2.000
    item rack-b weight 2.000
    item rack-c weight 2.000
}

# Replicated rule: 3 replicas across 3 racks (FirstN)
rule replicated_rack {
    id 0
    type replicated
    step take default
    step chooseleaf firstn 0 type rack
    step emit
}

# Erasure-coded rule: k=4, m=2 across 6 hosts (Indep keeps shard slots stable)
rule erasure_host {
    id 1
    type erasure
    step set_chooseleaf_tries 5
    step set_choose_tries 100
    step take default
    step chooseleaf indep 0 type host
    step emit
}
